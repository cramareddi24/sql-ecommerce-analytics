## 01_monthly_revenue_growth

**Question:** How is net revenue trending month over month?

```sql
-- Question: How is net revenue trending month over month?
-- Skills: CTE, date grouping, LAG window function
-- Net revenue = completed orders only, after discounts.

WITH monthly AS (
    SELECT
        strftime('%Y-%m', o.order_date)                          AS month,
        COUNT(DISTINCT o.order_id)                               AS orders,
        ROUND(SUM(oi.quantity * oi.unit_price * (1 - oi.discount)), 2) AS net_revenue
    FROM orders o
    JOIN order_items oi ON oi.order_id = o.order_id
    WHERE o.status = 'Completed'
    GROUP BY month
)
SELECT
    month,
    orders,
    net_revenue,
    LAG(net_revenue) OVER (ORDER BY month)                       AS prior_month_revenue,
    ROUND(100.0 * (net_revenue - LAG(net_revenue) OVER (ORDER BY month))
          / LAG(net_revenue) OVER (ORDER BY month), 1)           AS mom_growth_pct
FROM monthly
ORDER BY month;
```

## 03_category_revenue_share

**Question:** What share of revenue and profit does each category contribute?

<details><summary>SQL</summary>

```sql
-- Question: What share of revenue and profit does each category contribute?
-- Skills: window function over the full result set (SUM() OVER ())

WITH cat AS (
    SELECT
        p.category,
        SUM(oi.quantity * oi.unit_price * (1 - oi.discount))                 AS revenue,
        SUM(oi.quantity * (oi.unit_price * (1 - oi.discount) - p.unit_cost)) AS profit
    FROM order_items oi
    JOIN orders o   ON o.order_id = oi.order_id
    JOIN products p ON p.product_id = oi.product_id
    WHERE o.status = 'Completed'
    GROUP BY p.category
)
SELECT
    category,
    ROUND(revenue, 2)                                   AS net_revenue,
    ROUND(100.0 * revenue / SUM(revenue) OVER (), 1)    AS revenue_share_pct,
    ROUND(profit, 2)                                    AS gross_profit,
    ROUND(100.0 * profit / SUM(profit) OVER (), 1)      AS profit_share_pct
FROM cat
ORDER BY net_revenue DESC;
```

```sql
-- Question: Who are the top 3 customers in each region by lifetime spend?
-- Skills: RANK() with PARTITION BY, filtering on a window result

WITH spend AS (
    SELECT
        c.customer_id,
        c.first_name || ' ' || c.last_name                              AS customer,
        c.region,
        COUNT(DISTINCT o.order_id)                                      AS orders,
        ROUND(SUM(oi.quantity * oi.unit_price * (1 - oi.discount)), 2)  AS lifetime_spend
    FROM customers c
    JOIN orders o       ON o.customer_id = c.customer_id
    JOIN order_items oi ON oi.order_id = o.order_id
    WHERE o.status = 'Completed'
    GROUP BY c.customer_id
),
ranked AS (
    SELECT *, RANK() OVER (PARTITION BY region ORDER BY lifetime_spend DESC) AS region_rank
    FROM spend
)
SELECT region, region_rank, customer, orders, lifetime_spend
FROM ranked
WHERE region_rank <= 3
ORDER BY region, region_rank;
```

## 05_repeat_rate_by_acquisition_channel

```sql
-- Question: Which acquisition channels bring in customers who come back?
-- Skills: subquery aggregation, conditional counting with CASE

WITH per_customer AS (
    SELECT
        c.customer_id,
        c.acquisition_channel,
        COUNT(o.order_id) AS completed_orders
    FROM customers c
    LEFT JOIN orders o ON o.customer_id = c.customer_id AND o.status = 'Completed'
    GROUP BY c.customer_id
)
SELECT
    acquisition_channel,
    COUNT(*)                                                             AS customers,
    SUM(CASE WHEN completed_orders >= 2 THEN 1 ELSE 0 END)               AS repeat_customers,
    ROUND(100.0 * SUM(CASE WHEN completed_orders >= 2 THEN 1 ELSE 0 END) / COUNT(*), 1) AS repeat_rate_pct,
    ROUND(AVG(completed_orders), 2)                                      AS avg_orders_per_customer
FROM per_customer
GROUP BY acquisition_channel
ORDER BY repeat_rate_pct DESC;
```

## 06_cohort_retention

**Question:** Of customers who first ordered in a given quarter, how many ordered again later?

<details><summary>SQL</summary>

```sql
-- Question: Of customers who first ordered in a given quarter, how many ordered again later?
-- Skills: cohort analysis, month arithmetic, multiple CTEs

WITH first_order AS (
    SELECT customer_id, MIN(order_date) AS first_date
    FROM orders
    WHERE status <> 'Canceled'
    GROUP BY customer_id
),
activity AS (
    SELECT
        f.customer_id,
        strftime('%Y', f.first_date) || '-Q' || ((CAST(strftime('%m', f.first_date) AS INTEGER) + 2) / 3) AS cohort,
        (CAST(strftime('%Y', o.order_date) AS INTEGER) * 12 + CAST(strftime('%m', o.order_date) AS INTEGER))
      - (CAST(strftime('%Y', f.first_date) AS INTEGER) * 12 + CAST(strftime('%m', f.first_date) AS INTEGER)) AS months_since_first
    FROM first_order f
    JOIN orders o ON o.customer_id = f.customer_id AND o.status <> 'Canceled'
)
SELECT
    cohort,
    COUNT(DISTINCT customer_id)                                                        AS cohort_size,
    ROUND(100.0 * COUNT(DISTINCT CASE WHEN months_since_first BETWEEN 1 AND 3 THEN customer_id END)
          / COUNT(DISTINCT customer_id), 1)                                            AS returned_within_3mo_pct,
    ROUND(100.0 * COUNT(DISTINCT CASE WHEN months_since_first BETWEEN 1 AND 6 THEN customer_id END)
          / COUNT(DISTINCT customer_id), 1)                                            AS returned_within_6mo_pct
FROM activity
GROUP BY cohort
ORDER BY cohort;
```

```sql
-- Question: Which categories have the highest return rates, and what does that cost?
-- Skills: CASE inside aggregates, HAVING

SELECT
    p.category,
    COUNT(DISTINCT o.order_id)                                                         AS orders_with_category,
    COUNT(DISTINCT CASE WHEN o.status = 'Returned' THEN o.order_id END)                AS returned_orders,
    ROUND(100.0 * COUNT(DISTINCT CASE WHEN o.status = 'Returned' THEN o.order_id END)
          / COUNT(DISTINCT o.order_id), 1)                                             AS return_rate_pct,
    ROUND(SUM(CASE WHEN o.status = 'Returned'
                   THEN oi.quantity * oi.unit_price * (1 - oi.discount) ELSE 0 END), 2) AS returned_revenue
FROM orders o
JOIN order_items oi ON oi.order_id = o.order_id
JOIN products p     ON p.product_id = oi.product_id
WHERE o.status IN ('Completed', 'Returned')
GROUP BY p.category
HAVING COUNT(DISTINCT o.order_id) >= 50
ORDER BY return_rate_pct DESC;
```

## 08_running_total_and_moving_average

```sql
-- Question: What's cumulative revenue year to date, and what's the 3-month trend?
-- Skills: running total with PARTITION BY, moving average with a window frame

WITH monthly AS (
    SELECT
        strftime('%Y', o.order_date)                                 AS year,
        strftime('%Y-%m', o.order_date)                              AS month,
        SUM(oi.quantity * oi.unit_price * (1 - oi.discount))         AS revenue
    FROM orders o
    JOIN order_items oi ON oi.order_id = o.order_id
    WHERE o.status = 'Completed'
    GROUP BY month
)
SELECT
    month,
    ROUND(revenue, 2)                                                            AS net_revenue,
    ROUND(SUM(revenue) OVER (PARTITION BY year ORDER BY month), 2)               AS ytd_revenue,
    ROUND(AVG(revenue) OVER (ORDER BY month ROWS BETWEEN 2 PRECEDING AND CURRENT ROW), 2) AS moving_avg_3mo
FROM monthly
ORDER BY month;
```

## 09_aov_by_sales_channel

**Question:** Does average order value differ by sales channel, and is it changing?

<details><summary>SQL</summary>

```sql
-- Question: Does average order value differ by sales channel, and is it changing?
-- Skills: two-level aggregation (order level, then channel level), pivot with CASE

WITH order_totals AS (
    SELECT
        o.order_id,
        o.sales_channel,
        strftime('%Y', o.order_date)                             AS year,
        SUM(oi.quantity * oi.unit_price * (1 - oi.discount))     AS order_value
    FROM orders o
    JOIN order_items oi ON oi.order_id = o.order_id
    WHERE o.status = 'Completed'
    GROUP BY o.order_id
)
SELECT
    sales_channel,
    COUNT(*)                                                          AS orders,
    ROUND(AVG(order_value), 2)                                        AS avg_order_value,
    ROUND(AVG(CASE WHEN year = '2024' THEN order_value END), 2)       AS aov_2024,
    ROUND(AVG(CASE WHEN year = '2025' THEN order_value END), 2)       AS aov_2025_h1
FROM order_totals
GROUP BY sales_channel
ORDER BY avg_order_value DESC;
```
